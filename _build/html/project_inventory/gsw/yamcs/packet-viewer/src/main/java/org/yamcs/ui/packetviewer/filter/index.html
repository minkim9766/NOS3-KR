            if (!matchUnaryExpression(clause, packet)) {
                return false;
            }

        return true;

        boolean res;
        if (expression.getComparison() != null) {

            value = Integer.toString(packet.getLength());

                parameter = packet.getParameterForShortName(expression.ref);

            if (parameter == null) {
